# 启动器优化 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 PowerShell WinForms 启动器不再卡顿（状态轮询去子进程，3.1s→毫秒级）并升级为 Win11 深色亚克力外观，附带启动后自动开浏览器、打开项目目录按钮、隐藏控制台窗口。

**Architecture:** 性能核心是 `launcher.ps1` 的 1.5s 定时器不再 `& check-launcher.ps1` 拉子进程（单次约 3.1s，阻塞 UI 线程），改为直接调用已点源的 `launcher-core.ps1` 纯函数（`Load-PidFile`/`Is-ProcessAlive` 毫秒级）；端口检查只在点"启动"时做。外观通过 P/Invoke `dwmapi.dll` 设深色标题栏（attr 20）+ 亚克力背景（attr 38），失败逐级降级到云母→纯色；按钮用 `FlatStyle.Flat` + Paint 自绘圆角。`launcher-core.ps1` 零改动。

**Tech Stack:** Windows PowerShell 5.1 / WinForms / System.Drawing / dwmapi P/Invoke / 批处理

**规格:** `docs/superpowers/specs/2026-08-06-启动器优化-design.md`

function Draw-RoundedRect($g, [System.Drawing.Rectangle]$rect, [System.Drawing.Color]$color, [int]$radius) {
    $d = $radius * 2
    $path = New-Object System.Drawing.Drawing2D.GraphicsPath
    $path.AddArc($rect.X, $rect.Y, $d, $d, 180, 90)
    $path.AddArc($rect.Right - $d, $rect.Y, $d, $d, 270, 90)
    $path.AddArc($rect.Right - $d, $rect.Bottom - $d, $d, $d, 0, 90)
    $path.AddArc($rect.X, $rect.Bottom - $d, $d, $d, 90, 90)
    $path.CloseFigure()
    $g.SmoothingMode = [System.Drawing.Drawing2D.SmoothingMode]::AntiAlias
    $brush = New-Object System.Drawing.SolidBrush($color)
    $g.FillPath($brush, $path)
    $brush.Dispose()
    $path.Dispose()
}

function Set-StatusText([string]$text, [System.Drawing.Color]$color) {
    if ($script:LastStatusText -eq $text) { return }
    $script:LastStatusText = $text
    $lblStatus.Text = $text
    $lblStatus.ForeColor = $color
}

# 性能关键：全部走进程内 core 函数（毫秒级），绝不拉 check-launcher.ps1 子进程
function Update-Status {
    if ($script:SuppressStatus) { return }
    $pidValue = Load-PidFile
    $alive = $false
    if ($pidValue) { $alive = Is-ProcessAlive $pidValue }
    if ($alive) {
        Set-StatusText ("● 运行中   PID " + $pidValue + "   " + (Get-ServerUrl)) $script:ColGreen
        if ($script:PendingOpenBrowser) {
            $script:PendingOpenBrowser = $false
            Start-Process (Get-ServerUrl)
        }
    } elseif ($script:PendingOpenBrowser) {
        Set-StatusText "● 正在启动…" $script:ColAmber
    } else {
        Set-StatusText "● 已停止" $script:ColGray
    }
}

function Read-Logs {
    $out = Get-DevOutPath
    $err = Get-DevErrPath
    $text = ""
    if (Test-Path $out) { $text += (Get-Content $out -Tail 30 -ErrorAction SilentlyContinue) -join "`r`n" }
    if (Test-Path $err) {
        $errText = (Get-Content $err -Tail 10 -ErrorAction SilentlyContinue) -join "`r`n"
        if ($errText.Trim()) { $text += "`r`n--- stderr ---`r`n" + $errText }
    }
    $newText = "暂无日志"
    if ($text.Trim()) { $newText = $text }
    if ($txtLog.Text -ne $newText) {
        $txtLog.Text = $newText
        $txtLog.SelectionStart = $txtLog.TextLength
        $txtLog.ScrollToCaret()
    }
}
